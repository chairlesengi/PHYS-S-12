#include <Adafruit_NeoPixel.h>

#define HALL_PIN A0
#define LED_PIN D6

Adafruit_NeoPixel out(11,LED_PIN, NEO_GRB + NEO_KHZ800);

void colorStrip(int r, int g, int b){
  for(int i = 0; i < out.numPixels(); i++){
    out.setPixelColor(i, out.Color(r,g,b));
  }
  out.show();
}

void setup() {
  // put your setup code here, to run once:
  pinMode(HALL_PIN, INPUT);
  Serial.begin(9600);
  out.begin();
  out.show();
  out.setBrightness(255);
}

void loop() {
  // put your main code here, to run repeatedly:
  int output = map(analogRead(HALL_PIN)-1790,0,1790,0,255);
  if(output < 0){
    colorStrip(0,-1*output,-1*output);
  } else{
    colorStrip(output,0,output);
  }
  Serial.println(output);

}
